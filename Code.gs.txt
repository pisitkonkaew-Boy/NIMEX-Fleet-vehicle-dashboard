/** NIMEX Fleet Vehicle — Apps Script (วางใน Extensions > Apps Script ของ Google Sheet ใหม่ บันทึกเป็น Code.gs)
 *  Script Properties: ADMIN_KEY = รหัสผู้ดูแล (ห้ามเขียนรหัสลงไฟล์นี้)
 *  Deploy > New deployment > Web app > Execute as: Me · Who has access: Anyone → คัดลอก URL /exec ใส่ใน config.js
 *  แท็บ Rows, Upload (พักข้อมูลระหว่างอัปโหลด), Exempt และ Log สร้างให้อัตโนมัติ
 *  อัปโหลด: stage_begin → stage_append (ทีละ ≤3,000 แถว ลงแท็บ Upload) → stage_commit (รวมเข้า Rows ครั้งเดียว: เลขที่ใบรายการซ้ำ = ใช้ค่าใหม่, ใหม่ = ต่อท้าย)
 *  ถ้าอัปโหลดขาดกลางทาง แท็บ Rows ไม่ถูกแตะ
 *  อัปเดตโค้ดนี้แล้ว: Deploy > Manage deployments > ✏️ แก้ไข > Version: New version > Deploy (ลิงก์ /exec เดิมใช้ต่อได้) */
var COLS = ['tripId','departDate','returnDate','plate','vehicleType','dc','releaseType','distance','rawDistance',
  'mileOut','mileIn','dayStatus','mileDiff','tripStatus','route'];
var TEXT_COLS = ['tripId','departDate','returnDate','plate','vehicleType','dc','releaseType','dayStatus','tripStatus','route'];
var CK = 'fv-data', CHUNK = 90000;

function ss_() { return SpreadsheetApp.getActiveSpreadsheet(); }
function tab_(name, head) {
  var t = ss_().getSheetByName(name);
  if (!t) { t = ss_().insertSheet(name); t.getRange(1, 1, 1, head.length).setValues([head]).setFontWeight('bold'); t.setFrozenRows(1); }
  return t;
}
function rowsTab_() { return tab_('Rows', COLS); }
function upTab_() { return tab_('Upload', COLS); }
function cleanRows_(rows) { return rows.map(function (r) { return COLS.map(function (c, i) { var v = r[i]; return v === null || v === undefined ? '' : v; }); }); }
function writeRows_(t, start, rows) {
  if (!rows.length) return;
  TEXT_COLS.forEach(function (c) { t.getRange(start, COLS.indexOf(c) + 1, rows.length, 1).setNumberFormat('@'); });
  t.getRange(start, 1, rows.length, COLS.length).setValues(rows);
}
function readRows_(t) { var n = t.getLastRow(); return n < 2 ? [] : t.getRange(2, 1, n - 1, COLS.length).getValues(); }
var EX_HEAD = ['plate', 'note'];
function exTab_() { return tab_('Exempt', EX_HEAD); }
function settings_() {
  var t = exTab_(), n = t.getLastRow(), ex = [];
  if (n > 1) t.getRange(2, 1, n - 1, 2).getValues().forEach(function (r) { if (String(r[0]).trim()) ex.push({ plate: String(r[0]).trim(), note: String(r[1] || '') }); });
  var th = null; try { th = JSON.parse(PropertiesService.getScriptProperties().getProperty('THRESHOLDS') || 'null'); } catch (e) {}
  var gr = []; try { gr = JSON.parse(PropertiesService.getScriptProperties().getProperty('GRADES') || '[]'); } catch (e) {}
  return { exempt: ex, thresholds: th, grades: gr };
}
function log_(a, d) { tab_('Log', ['time', 'action', 'detail']).appendRow([new Date(), a, d]); }
function json_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }

function publicJson_() {
  var t = rowsTab_(), n = t.getLastRow();
  var rows = n < 2 ? [] : t.getRange(2, 1, n - 1, COLS.length).getValues();
  return JSON.stringify({ ok: true, cols: COLS, rows: rows, settings: settings_(), updated: PropertiesService.getScriptProperties().getProperty('UPDATED') || '' });
}
function cacheGet_() {
  var c = CacheService.getScriptCache(), n = c.get(CK + '-n'); if (!n) return null;
  var keys = []; for (var i = 0; i < +n; i++) keys.push(CK + '-' + i);
  var all = c.getAll(keys), b = '';
  for (i = 0; i < keys.length; i++) { if (!all[keys[i]]) return null; b += all[keys[i]]; }
  return Utilities.ungzip(Utilities.newBlob(Utilities.base64Decode(b), 'application/x-gzip')).getDataAsString('UTF-8');
}
function cachePut_(s) {
  var b = Utilities.base64Encode(Utilities.gzip(Utilities.newBlob(s, 'text/plain')).getBytes()), o = {}, n = 0;
  for (var i = 0; i < b.length; i += CHUNK) o[CK + '-' + (n++)] = b.substr(i, CHUNK);
  if (n <= 90) { o[CK + '-n'] = String(n); CacheService.getScriptCache().putAll(o, 21600); }
}
function cacheClear_() { CacheService.getScriptCache().remove(CK + '-n'); }

function doGet() {
  var s = null;
  try { s = cacheGet_(); } catch (e) {}
  if (!s) { s = publicJson_(); try { cachePut_(s); } catch (e) {} }
  return ContentService.createTextOutput(s).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var b;
  try { b = JSON.parse(e.postData.contents); } catch (x) { return json_({ ok: false, error: 'bad_request' }); }
  var key = PropertiesService.getScriptProperties().getProperty('ADMIN_KEY');
  if (!key) return json_({ ok: false, error: 'admin_key_not_set' });
  if (!b.key || b.key !== key) { Utilities.sleep(1200); return json_({ ok: false, error: 'denied' }); }
  if (b.a === 'auth') return json_({ ok: true });
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(25000)) return json_({ ok: false, error: 'busy' });
  try {
    if (b.a === 'settings') {           // ทะเบียนยกเว้น (แท็บ Exempt) + เกณฑ์แจ้งเตือน
      var st = b.settings || {}, ex = Array.isArray(st.exempt) ? st.exempt : [];
      if (ex.length > 2000) return json_({ ok: false, error: 'too_many' });
      var et = exTab_(), en = et.getLastRow();
      if (en > 1) et.getRange(2, 1, en - 1, 2).clearContent();
      if (ex.length) {
        et.getRange(2, 1, ex.length, 2).setNumberFormat('@').setValues(ex.map(function (e) { return [String(e.plate || '').slice(0, 40), String(e.note || '').slice(0, 200)]; }));
      }
      if (st.thresholds) PropertiesService.getScriptProperties().setProperty('THRESHOLDS', JSON.stringify(st.thresholds).slice(0, 2000));
      if (Array.isArray(st.grades)) { var gj = JSON.stringify(st.grades); if (gj.length > 8000) return json_({ ok: false, error: 'grades_too_long' }); PropertiesService.getScriptProperties().setProperty('GRADES', gj); }
      log_('settings', ex.length + ' exempt plates');
      cacheClear_(); try { cachePut_(publicJson_()); } catch (x) {}
      return json_({ ok: true });
    }
    if (b.a === 'stage_begin') {         // เริ่มอัปโหลดใหม่: ล้างแท็บพัก (Rows ไม่ถูกแตะ)
      var ut = upTab_(), un = ut.getLastRow(); if (un > 1) ut.getRange(2, 1, un - 1, COLS.length).clearContent();
      return json_({ ok: true });
    }
    if (b.a === 'stage_append') {
      var sr = b.rows || [];
      if (!Array.isArray(sr) || sr.length > 3000) return json_({ ok: false, error: 'bad_rows' });
      var u2 = upTab_(); writeRows_(u2, u2.getLastRow() + 1, cleanRows_(sr));
      return json_({ ok: true, staged: u2.getLastRow() - 1 });
    }
    if (b.a === 'stage_commit') {        // รวมแท็บพักเข้า Rows ในครั้งเดียว · replace=true แทนที่ทั้งหมด (นำเข้าครั้งแรก)
      var up = readRows_(upTab_());
      if (b.expect && up.length !== b.expect) return json_({ ok: false, error: 'staged_' + up.length + '_of_' + b.expect });
      var rt = rowsTab_(), cur = b.replace ? [] : readRows_(rt), idx = {}, added = 0, updated = 0;
      cur.forEach(function (r, i) { idx[String(r[0])] = i; });
      up.forEach(function (r) { var k = String(r[0]); if (k in idx) { cur[idx[k]] = r; updated++; } else { idx[k] = cur.length; cur.push(r); added++; } });
      var old = rt.getLastRow() - 1;
      writeRows_(rt, 2, cur);
      if (old > cur.length) rt.getRange(cur.length + 2, 1, old - cur.length, COLS.length).clearContent();
      var uc = upTab_(), ucn = uc.getLastRow(); if (ucn > 1) uc.getRange(2, 1, ucn - 1, COLS.length).clearContent();
      PropertiesService.getScriptProperties().setProperty('UPDATED', new Date().toISOString());
      log_(b.note || 'upload', 'new ' + added + ' · updated ' + updated + ' · total ' + cur.length);
      cacheClear_(); try { cachePut_(publicJson_()); } catch (x) {}
      return json_({ ok: true, total: cur.length, added: added, updated: updated });
    }
    var t = rowsTab_();
    if (b.a === 'begin') {               // replace everything: an upload replaces all trips (same as the original page)
      var n = t.getLastRow(); if (n > 1) t.getRange(2, 1, n - 1, COLS.length).clearContent();
      return json_({ ok: true });
    }
    if (b.a === 'append') {              // ~2,500 rows per call, in COLS order
      var rows = b.rows || [];
      if (!Array.isArray(rows) || rows.length > 3000) return json_({ ok: false, error: 'bad_rows' });
      if (!rows.length) return json_({ ok: true });
      var start = t.getLastRow() + 1;
      TEXT_COLS.forEach(function (c) { t.getRange(start, COLS.indexOf(c) + 1, rows.length, 1).setNumberFormat('@'); });
      t.getRange(start, 1, rows.length, COLS.length).setValues(rows.map(function (r) {
        return COLS.map(function (c, i) { var v = r[i]; return v === null || v === undefined ? '' : v; });
      }));
      return json_({ ok: true, total: t.getLastRow() - 1 });
    }
    if (b.a === 'finish') {
      var total = t.getLastRow() - 1;
      PropertiesService.getScriptProperties().setProperty('UPDATED', new Date().toISOString());
      log_(b.note || 'upload', total + ' rows');
      cacheClear_(); try { cachePut_(publicJson_()); } catch (x) {}
      return json_({ ok: true, total: total });
    }
    return json_({ ok: false, error: 'unknown' });
  } finally { lock.releaseLock(); }
}
